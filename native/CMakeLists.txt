cmake_minimum_required(VERSION 3.24)
project(CelesteGraphics LANGUAGES C CXX)

if(APPLE)
    set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
    set(CMAKE_INSTALL_RPATH "@loader_path")
    set(CMAKE_INSTALL_NAME_DIR "@rpath")
    set(CMAKE_BUILD_WITH_INSTALL_NAME_DIR ON)
elseif(UNIX)
    set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
    set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()

option(CELESTE_BUILD_EFFECTS "Build the shared effect frontend" ON)
option(CELESTE_BUILD_DIRECT3D11 "Build the Direct3D11 backend" OFF)
option(CELESTE_BUILD_DIRECT3D12 "Build the Direct3D12 backend" OFF)
option(CELESTE_BUILD_VULKAN "Build the Vulkan backend" OFF)

if(CELESTE_BUILD_EFFECTS)
    add_subdirectory(Effects)
endif()
if(CELESTE_BUILD_DIRECT3D11)
    add_subdirectory(Graphics/Direct3D11)
endif()
if(CELESTE_BUILD_VULKAN)
    add_subdirectory(Graphics/Vulkan)
endif()
if(CELESTE_BUILD_DIRECT3D12)
    add_subdirectory(Graphics/Direct3D12)
endif()
